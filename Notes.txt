Startup for cpp connector:
path: sudo nano /etc/systemd/system/myapp.service
sudo systemctl daemon-reload
sudo systemctl enable myscript.service
sudo systemctl start myscript.service


Fra filen myapp.service (ligger som fil i aktiv mappe):

[Unit]
Description=cpp database connector start
After=network.target

[Service]
# Replace 'pi' with your actual username
User=oliver
# Use the full path to your executable
ExecStart=/home/oliver/code/Database_Connector_mqttPAHO_cmake/out/build/mqtt_sub
# Restart the app automatically if it crashes
Restart=always
# Time to wait before restarting
RestartSec=5

[Install]
WantedBy=multi-user.target
